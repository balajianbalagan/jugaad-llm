Drop-In Compatibility

Zero Refactoring Required.

jugaad-llm implements the full OpenAI Chat Completions specification with SSE token streaming.

from openai import OpenAI

# 🧙‍♂️ Point directly to your local or ngrok jugaad-llm gateway
client = OpenAI(
    base_url="http://localhost:4000/v1",  # Or https://your-tunnel.ngrok-free.app/v1
    api_key="sk-hackathon-token"          # Master gateway key from .env
)

response = client.chat.completions.create(
    model="auto",                         # Or "chatgpt", "claude", "deepseek", "grok"
    messages=[
        {"role": "system", "content": "You are an elite hackathon pair programmer."},
        {"role": "user", "content": "Architect an autonomous multi-agent research workflow."}
    ],
    stream=True
)

for chunk in response:
    content = chunk.choices[0].delta.content or ""
    print(content, end="", flush=True)
import { createOpenAI } from '@ai-sdk/openai';
import { streamText } from 'ai';

// 🧙‍♂️ Drop-in OpenAI provider pointed to jugaad-llm
const jugaad = createOpenAI({
  baseURL: 'http://localhost:4000/v1',
  apiKey: 'sk-hackathon-token',
});

const result = await streamText({
  model: jugaad('claude'), // routes seamlessly to your Claude free web session!
  prompt: 'Generate 3 high-converting landing page hooks for our AI product.',
});

for await (const delta of result.textStream) {
  process.stdout.write(delta);
}
